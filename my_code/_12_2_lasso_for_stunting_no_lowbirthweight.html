# 2. 各変数の年次変動を評価する関数
# ******************************************************************************
#' @title calculate_temporal_stability
#' @description 指定された変数について、年別の平均値を計算し、変動係数（CV）を用いて時間不変性を評価します。
#' @param df データフレーム
#' @param var_names 評価する変数名のベクトル
#' @param year_col 年を示す列名（デフォルトは "year"）
#' @return 各変数の平均値、標準偏差、変動係数、範囲ベースの変動指標、時間安定性の分類を含むデータフレーム
# ******************************************************************************
calculate_temporal_stability <- function(df, var_names, year_col = "year") {
  
  # 年別平均値を計算
  yearly_means <- df %>%
    group_by(!!sym(year_col)) %>%
    summarise(across(all_of(var_names), ~ mean(.x, na.rm = TRUE)), .groups = "drop")
  
  # 各変数の変動係数（CV）と時間安定性指標を計算
  stability_results <- map_dfr(var_names, function(var) {
    yearly_values <- yearly_means[[var]]
    
    # 基本統計量
    mean_val <- mean(yearly_values, na.rm = TRUE)
    sd_val <- sd(yearly_values, na.rm = TRUE)
    cv_val <- sd_val / mean_val
    
    # 範囲ベースの変動指標
    min_val <- min(yearly_values, na.rm = TRUE)
    max_val <- max(yearly_values, na.rm = TRUE)
    range_ratio <- (max_val - min_val) / mean_val
    
    # 時間安定性の分類
    stability_category <- case_when(
      cv_val < 0.05 ~ "stabilit:very high",
      cv_val < 0.10 ~ "stabilit: high", 
      cv_val < 0.20 ~ "stability: moderate",
      cv_val < 0.50 ~ "stability: low",
      TRUE ~ "unstable"
    )
    
    tibble(
      variable = var,
      mean_across_years = mean_val,
      sd_across_years = sd_val,
      cv_coefficient = cv_val,
      range_ratio = range_ratio,
      stability_category = stability_category
    )
  })
  
  # CV昇順でソート
  stability_results <- stability_results %>%
    arrange(cv_coefficient)
  
  return(stability_results)
}
# --- 関数定義ここまで ---

# 3. 関数を用いて選択された変数の時間安定性を評価

# 2. sampling weightの年次補正
# 年ごとの元 v005 合計を算出
year_totals <- df_lasso %>%
  group_by(year) %>%
  summarise(total_v005 = sum(v005, na.rm = TRUE), .groups = 'drop')

# 正規化済みサンプリングウェイトの作成
df_lasso <- df_lasso %>%
  select(-any_of("total_v005")) %>%  # 存在する場合のみ削除
  left_join(year_totals, by = "year") %>%
  mutate(
    # 元のDHSウェイト
    dhs_weight = v005 / 1e6,
    # 年次正規化係数：全期間対象の標本サイズ比 (例：年の全体ウェイト比)
    year_norm = total_v005 / sum(year_totals$total_v005),
    # 正規化済みウェイト
    sampling_weight_norm = dhs_weight / year_norm,
    # 極端値トリミング
    sampling_weight_trimmed = pmax(
      pmin(sampling_weight_norm,
           quantile(sampling_weight_norm, 0.99, na.rm = TRUE)),
      quantile(sampling_weight_norm, 0.01, na.rm = TRUE)
    )
  )

# 3. surveydesignオブジェクトの作成
# Survey designオブジェクトの作成

  ids = ~v021,          # Primary Sampling Unit (クラスター)

  weights = ~sampling_weight_trimmed,
