save_path <- file.path(gdrive_dir, "output")

# ステップ1.1: all_data_merged.rdsの読み込み
df_analysis <- readRDS(file.path(save_path, "all_data_merged.rds"))

# haven_labelled → numeric へ変換
df_analysis <- df_analysis %>%
  mutate(
    treatment_group = as.numeric(unclass(treatment_group))
    # 必要なら他のラベル付き変数もここで変換
    # across(c(treatment_group, ・・・), ~ as.numeric(unclass(.x)))
  )

# ステップ1.2:  初回導入年Gの作成
#' Repeated cross section では個体が入れ替わるため、真の panel ID は存在しません。
#' ただし att_gt() は panel = FALSE の場合でも idname を形式的に要求します。
#' そこで仮想 ID を作成します。

df_analysis <- df_analysis %>%
  mutate(
   # 初回導入年 g の作成
    g = case_when(
      treatment_group == 1 ~ 2010L,  # treat_early(2010)
      treatment_group == 2 ~ 2012L,  # treat_late(2012)
      treatment_group == 0 ~ 0L,     # control (never treated)
      TRUE ~ NA_integer_
    ),
    # 仮想 ID（repeated cross section）
    id = as.integer(row_number()),
    # 型の明示的な変換
    year = as.integer(year),
    g = as.integer(g),
    nt_ch_stunt = as.numeric(nt_ch_stunt),
    sampling_weight_trimmed = as.numeric(sampling_weight_trimmed),
    v021 = as.character(v021)  # クラスター変数は文字列でOK
  )
