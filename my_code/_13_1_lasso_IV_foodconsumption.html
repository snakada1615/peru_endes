#' @title create_lasso_data
#' @description LASSO用のデータセットを作成する関数
#' @param df 元データフレーム
#' @param drop_cols_init 除外変数定義(初期値)
#' @param drop_cols_add 除外変数定義(追加)
#' @param grouping_cols ID・管理用変数
#' @param y_var_all 目的変数候補
#' @param y_var 目的変数
#' @return リスト(df_lasso, covariate_names)
# *******************************************************************************
create_lasso_data <- function(df, 
          drop_cols_init, 
          drop_cols_add, 
          grouping_cols,
          y_var_all,
          y_var) {

  # 除外変数定義(最終値)
  drop_cols <- c(drop_cols_init, y_var_all, drop_cols_add) %>% unique()
  y_var_drop <- setdiff(y_var_all, y_var)

  
  # 1. LASSO用のデータセット定義
  df_lasso <- df
  
  cat("介入前データの群別サンプル数:\n")
  print(table(df_lasso$treatment_group))
  
  # 4. NAの多い列を除去（10%以上NA）
  initial_cols <- ncol(df_lasso)
  keep_cols <- names(df_lasso)[sapply(df_lasso, function(x) sum(is.na(x)) <= nrow(df_lasso) * 0.1)]
  df_lasso <- df_lasso[, keep_cols]
  cat("NA値の多い列を", initial_cols - ncol(df_lasso), "個削除しました\n")
  
  # 5. 共変量名の再定義（NA除去後）
  all_drop_cols <- intersect(drop_cols, names(df_lasso))
  grouping_cols <- intersect(grouping_cols, names(df_lasso))
  covariate_names <- setdiff(names(df_lasso), c(all_drop_cols, grouping_cols))
  
  
  # 6. 共変量のNAがある行を除去
  # df_lasso <- df_lasso[, covariate_names]
  complete_idx <- complete.cases(df_lasso)
  initial_rows <- nrow(df_lasso)
  df_lasso <- df_lasso[complete_idx, ]
  cat("NA値を含む行を", initial_rows - nrow(df_lasso), "行削除しました\n")
  cat("最終的な介入前データサイズ:", nrow(df_lasso), "行 ×", ncol(df_lasso), "列\n\n")
  
  return(list(df_lasso = df_lasso, covariate_names = covariate_names))
}

# 年ごとの元 v005 合計を算出
year_totals <- df_lasso %>%
  group_by(year) %>%
  summarise(total_v005 = sum(v005, na.rm = TRUE), .groups = 'drop')

# Survey designオブジェクトの作成前に変数を変換
df_lasso <- df_lasso %>%
  mutate(
    v021 = as.numeric(v021),                        # Primary Sampling Unit
    v022 = as.character(haven::as_factor(v022))    #haven_labelledをfactorに変換してからcharacter
  )

# 正規化済みサンプリングウェイトの作成
df_lasso <- df_lasso %>%
  dplyr::select(-any_of(c("total_v005"))) %>% # 存在する場合のみ削除
  left_join(year_totals, by = "year") %>%
  mutate(
    # 元のDHSウェイト
    dhs_weight = v005 / 1e6,
    # 年次正規化係数：全期間対象の標本サイズ比 (例：年の全体ウェイト比)
    year_norm = total_v005 / sum(year_totals$total_v005),
    # 正規化済みウェイト
    sampling_weight_norm = dhs_weight / year_norm,
    # 極端値トリミング
    sampling_weight_trimmed = pmax(
      pmin(sampling_weight_norm,
           quantile(sampling_weight_norm, 0.99, na.rm = TRUE)),
      quantile(sampling_weight_norm, 0.01, na.rm = TRUE)
    )
  )
