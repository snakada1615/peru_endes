# メディエータMに対するDID推定
#****************************************************************************
#' @title メディエータMに対するDID推定関数
#' @param mediator_variable メディエータ変数の名前（文字列）
#' @return メディエータに関するDID推定結果のリスト
#' @details
#' この関数は、指定されたメディエータ変数に対してD
#' ID推定を行い、集約効果や動的効果、グループ別効果、並行トレンド検定のp値を
#' 含むリストを返します。 
#' ****************************************************************************
att_mediator <- function(mediator_variable){
  att_mediator <- att_gt(
    yname   = mediator_variable,  # 例: re_anc_4vs
    tname   = "year",
    idname  = "id",
    gname   = "g",
    xformla = ~ 1,
    data    = df_analysis,
    panel   = FALSE,
    weightsname = "sampling_weight_trimmed",  # ← 修正：weightsname
    clustervars = "v021",
    control_group = "notyettreated",
    est_method = "dr",
    print_details = TRUE                     # ← 残してOK
  )
  
  # 集約効果の計算
  agg_simple_med <- aggte(att_mediator, type = "simple")
  agg_dynamic_med <- aggte(att_mediator, type = "dynamic")
  agg_group_med <- aggte(att_mediator, type = "group")

  # 結果の保存
  results_mediator <- list(
    mediator_name = mediator_variable,
    overall_att = agg_simple_med,
    dynamic_effects = agg_dynamic_med,
    group_effects = agg_group_med,
    parallel_trends_pvalue = summary(att_mediator)$DIDparams$pretest_pval
  )

return(results_mediator)
}
